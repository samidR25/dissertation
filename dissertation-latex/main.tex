% ── Convenience macros — keep clinical-honesty ordering consistent ────────
% Use \rawthen{raw}{corrected}{explanation} wherever a raw/uncorrected number
% and a corrected/honest number both exist for the same result, per the
% project's standing methodological-honesty rule: raw number stated in full,
% first, before the corrected figure is introduced.
\newcommand{\rawthen}[3]{%
    \textbf{Raw/uncorrected figure:} #1. #3\\
    \textbf{Corrected figure:} #2%
}

% Placeholder for a not-yet-inserted figure/image asset. Renders as a
% clearly labelled boxed placeholder in the compiled PDF so it is obvious
% at a glance what still needs inserting and what filename/caption to use.
% Usage: \figplaceholder{expected-filename.png}{Caption text}{fig:label}
\newcommand{\figplaceholder}[3]{%
    \begin{figure}[H]
        \centering
        \fbox{\parbox{0.85\textwidth}{\centering\vspace{1.5cm}
        \textbf{[INSERT FIGURE HERE]}\\[0.3cm]
        \texttt{\detokenize{#1}}
        \vspace{1.5cm}}}
        \caption{#2}
        \label{#3}
    \end{figure}%
}

% A real, already-generated figure. Usage:
% \realfig{filename.png}{width-fraction e.g. 0.8}{Caption text}{fig:label}
\newcommand{\realfig}[4]{%
    \begin{figure}[H]
        \centering
        \includegraphics[width=#2\textwidth]{#1}
        \caption{#3}
        \label{#4}
    \end{figure}%
}

% Same as \realfig, but rotated onto its own landscape page -- use for
% wide horizontal pipeline/block diagrams where portrait width squeezes
% the text illegibly small even at width=1.0\textwidth. Width fraction
% here is relative to \textheight (the rotated page's usable "width").
\newcommand{\realfigwide}[4]{%
    \begin{sidewaysfigure}
        \centering
        \includegraphics[width=#2\textheight]{#1}
        \caption{#3}
        \label{#4}
    \end{sidewaysfigure}%
}
